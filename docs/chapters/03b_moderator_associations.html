  ggplot2::theme_minimal(base_size = base_size) +
    ggplot2::theme(

cramers_v_corrected <- function(x, y) {
  tab <- table(x, y)
  n <- sum(tab)
  if (n < 2L || nrow(tab) < 2L || ncol(tab) < 2L) return(NA_real_)
  chi <- suppressWarnings(stats::chisq.test(tab, correct = FALSE))
  phi2 <- unname(chi$statistic) / n
  r <- nrow(tab)
  k <- ncol(tab)
  phi2_corrected <- max(0, phi2 - ((k - 1) * (r - 1)) / (n - 1))
  r_corrected <- r - ((r - 1)^2) / (n - 1)
  k_corrected <- k - ((k - 1)^2) / (n - 1)
  denom <- min(k_corrected - 1, r_corrected - 1)
  if (denom <= 0) NA_real_ else sqrt(phi2_corrected / denom)
}

eta_squared_test <- function(y, group) {
  keep <- stats::complete.cases(y, group)
  y <- y[keep]
  group <- droplevels(factor(group[keep]))
  if (length(y) < 3L || nlevels(group) < 2L) {
    return(tibble::tibble(estimate = NA_real_, statistic = NA_real_, p_value = NA_real_))
  }
  fit <- stats::lm(y ~ group)
  tab <- stats::anova(fit)
  tibble::tibble(
    estimate = tab$`Sum Sq`[1] / sum(tab$`Sum Sq`),
    statistic = tab$`F value`[1],
    p_value = tab$`Pr(>F)`[1]
  )
}

significance_mark <- function(p) {
  dplyr::case_when(
    is.na(p) ~ "",
    p < 0.001 ~ "***",
    p < 0.01 ~ "**",
    p < 0.05 ~ "*",
    TRUE ~ ""
  )
}

5.1 Purpose and interpretation

This chapter checks whether the moderators overlap strongly enough to complicate their separate interpretation. Association is measured on the scale appropriate to each pair: Pearson’s r for two continuous variables, bias-corrected Cramér’s V for two categorical variables, and the correlation ratio η² for a continuous variable grouped by a categorical variable. Values closer to one indicate stronger association; they do not imply causation.

The tests below treat contrasts as rows and are therefore descriptive diagnostics rather than independent-observation causal tests. The study, system, and phylogenetic dependence handled by the meta-analytic models is not represented in these simple pairwise tests. Holm-adjusted p-values control multiplicity within this chapter.

5.4 Continuous–categorical associations

For each mixed pair, η² is the proportion of variation in the continuous moderator explained by the categorical grouping in a one-way model. Because η² is non-negative, the heatmap shows strength but not direction; group distributions should be consulted when direction matters.

5.5 Reading the diagnostics

For screening purposes, effect-size magnitude is more informative than significance because the dataset contains thousands of contrasts. As rough descriptive guides, values near 0.1, 0.3, and 0.5 can be read as small, moderate, and large for r or Cramér’s V. For η², values near 0.01, 0.06, and 0.14 are often described as small, moderate, and large. These are conventions rather than decision thresholds; substantive overlap, cell counts, and the dependency structure should guide model specification.

## Continuous–categorical associations

For each mixed pair, η² is the proportion of variation in the continuous moderator explained by the categorical grouping in a one-way model. Because η² is non-negative, the heatmap shows strength but not direction; group distributions should be consulted when direction matters.

## Reading the diagnostics

For screening purposes, effect-size magnitude is more informative than significance because the dataset contains thousands of contrasts. As rough descriptive guides, values near 0.1, 0.3, and 0.5 can be read as small, moderate, and large for *r* or Cramér's *V*. For η², values near 0.01, 0.06, and 0.14 are often described as small, moderate, and large. These are conventions rather than decision thresholds; substantive overlap, cell counts, and the dependency structure should guide model specification.
