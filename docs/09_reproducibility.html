model_files <- list.files(
  here::here("outputs", "models"),
  pattern = "\\.rds$",
  recursive = TRUE,
  full.names = TRUE
)

if (length(model_files) > 0) {
  tibble::tibble(
    path     = model_files,
    filename = basename(model_files),
    size_mb  = round(file.size(model_files) / 1024^2, 2),
    modified = file.mtime(model_files)
  ) |>
    dplyr::arrange(modified)
} else {
  cat("No cached model files found in outputs/models/.\n")
}
